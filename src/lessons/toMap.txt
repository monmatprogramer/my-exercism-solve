parentTree = {A: [B, C, D]}

//Convert parentTree to Map
// parentGraph = {A: [B, C, D]} 
parentGraph = new Map<string,string[]>(Object.entries(parentTree))


//step 1
// parentGraph = {A: [B, C, D]}  
// Get child from parent A
IF parentGraph.size() === 1:
    // parentKey = A
    parentKey = parentGraph.keys();
    // childLen = 3
    childLen = parentGraph.get().length()
    //childValue = [B,C,D]
    childValue = parentGraph.get();
    FOR 0 < childLen:
        
        //i=0
        //tempValue = childValue.shift() = B (childValue = [C,D])
        //parentGraph = {B: [A, C, D]}
        //

        //i=1
        //childValue.shift() = C (childValue = [D]) ❌ (childValue = [B,D]) ✅
        //parentGraph = {C: [A, B,D]}
       
         if(tempValue){
            // tempValue = B
            // childValue = [B,C,D]
           childValue.unshift(tempValue);
           //Swab array => childValue [C,B,D]
           childValue[0], childValue[1] = [childValue[1], childValue[0]]
         }
        tempValue = childValue.shift();
        parentGraph.set(tempValue, [parentKey, childValue])
